12. 统一导航工具设计
────────────────────────────────────────────────────────────────────────────────
  新建：frontend/src/utils/navigation.ts

  import { isNavigableResourceType, type NavigableResourceType, type TravelResourceType } from '@/types/resource'

  /** 正整数校验 */
  export function isValidResourceId(id: unknown): id is number {
    return typeof id === 'number' && Number.isInteger(id) && id > 0
  }

  /** 构造资源详情页 URL。general_activity 返回 null。未知类型返回 null。 */
  export function getResourceDetailUrl(type: TravelResourceType, id: number | null | undefined): string | null {
    if (!isValidResourceId(id)) return null
    if (!isNavigableResourceType(type)) return null

    if (type === 'scenic_spot') {
      return `/pages/scenic/detail?id=${id}`
    }

    // hotel / restaurant / entertainment / shopping_mall → 通用详情页
    return `/pages/resource/detail?type=${type}&id=${id}`
  }

  /** 打开资源详情页。不 Toast（由调用方决定 UI）。 */
  export function openResourceDetail(type: TravelResourceType, id: number | null | undefined): void {
    const url = getResourceDetailUrl(type, id)
    if (!url) return
    uni.navigateTo({ url })
  }

  函数职责分离：
    getResourceDetailUrl: 纯函数，URL 构造 + 校验
    openResourceDetail: 导航执行（调用 uni.navigateTo）

  不包含：API 请求、Toast、收藏评论、Transit

16. ResourceDetailViewModel 设计
────────────────────────────────────────────────────────────────────────────────

  决策：轻量 ViewModel adapter（方案 B）
  不把所有实体压成最小公共字段，保留实体差异。

  在通用详情页内定义 adapter 函数：

  // 公共基础字段
  interface ResourceBaseFields {
    name: string
    imageUrl: string | null
    description: string | null
    address: string | null
    score: number | null
    latitude: number | null
    longitude: number | null
  }

  // Hotel 独有
  function adaptHotel(h: Hotel): ResourceBaseFields & {
    priceText: string | null     // "¥300"
    openTime: string | null
    categoryText: null           // Hotel 无 category
  }

  // Restaurant 独有
  function adaptRestaurant(r: Restaurant): ResourceBaseFields & {
    priceText: string | null     // "人均 ¥50" 或 price_level 原文
    openTime: null               // Restaurant 无 open_time
    categoryText: string | null  // 菜系
  }

  adapter 只做格式化和字段重命名，不丢失实体独有字段。

  模板通过 v-if/v-else-if 按 resourceType 分支渲染差异字段。
  不对所有实体使用同一个循环渲染（避免过度抽象）。
